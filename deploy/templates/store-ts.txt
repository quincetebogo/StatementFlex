		name: ``
	};

	// Instance
	private static instance: ELEMENT_TYPEELEMENT_CAMEL_CASE_NAME;

	// --------------
	// INITIALISATION
	// --------------

	constructor() {

		super({
			name: `ELEMENT_TYPEELEMENT_PASCAL_CASE_NAME`,
			logStateChanges: STORE_LOG,
			persistence: `STORE_PERSISTANCE`
		});

		// Assignes Initial Store State or adds new properties
		if (!this.getState()) {
			this.setState(this.initialState, `INITIAL_STATE`);
		} else {
			const state: any = this.getState();

			Object.keys(this.initialState).forEach((propertyName) => {
				if (!Object.prototype.hasOwnProperty.call(state, propertyName)) {
					state[propertyName] = this.initialState[propertyName];
				}
			});
			this.setState(state, `UPDATE_INITIAL_STATE`);
		}
	}

	// --------------
	// EVENT HANDLERS

	}

	// ---------------
	// PROPERTIES
	// ---------------

	set name(value: string) {
		const _newState = { name: value };

		this.setState(_newState, `SET_NAME`);
	}

	get name(): string {
		return this.getStateProperty(`name`) as string || this.initialState.name;
	}
